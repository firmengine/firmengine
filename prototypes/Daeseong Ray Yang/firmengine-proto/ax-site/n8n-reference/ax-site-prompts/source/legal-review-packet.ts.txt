export interface LegalReviewPacket {payload:Record<string,any>;text:string;providedFactIds:string[];coverage:Record<string,any>}
const text=(value:unknown,max=500)=>String(value&&typeof value==="object"?JSON.stringify(value):value??"").slice(0,max);
const normalizedKey=(key:string)=>key.trim().replace(/([a-z0-9])([A-Z])/g,"$1_$2").replace(/\[\]/g,"").toLowerCase();
const axis=(key:string)=>normalizedKey(key).split(".")[0].replace(/_source$/g,"").replace(/s$/g,"");
const priority=(fact:Fact)=>fact.state==="verified"&&fact.recognition==="accepted"?0:fact.state==="verified"?1:fact.state==="extracted"?2:3;
function featureMatchTier(feature:string,fact:Fact):0|1|2|null{const wanted=normalizedKey(feature),actual=normalizedKey(fact.key);if(actual===wanted)return 0;if(axis(feature)===axis(fact.key)&&actual.split(".").at(-1)===wanted.split(".").at(-1))return 1;if(axis(feature)===axis(fact.key))return 2;return null;}
function factMatches(feature:string,fact:Fact){return featureMatchTier(feature,fact)!==null;}
function clip(value:unknown,max:number,path:string,truncated:string[]):unknown{if(typeof value!=="string")return value;return value.length>max?(truncated.push(path),value.slice(0,max)):value;}
function compactFact(fact:Fact,data:CaseData){const truncated:string[]=[];if(fact.refs.length>1)truncated.push("refs");if((fact.structuredRefs?.length??0)>1)truncated.push("structuredRefs");let value:any=fact.value;if(typeof value==="string")value=clip(value,600,"value",truncated);else if(value&&typeof value==="object"){const json=JSON.stringify(value);value=json.length>600?(truncated.push("value"),json.slice(0,600)):value;}const numeric=fact.numeric?{rawValue:clip(fact.numeric.rawValue,100,"numeric.rawValue",truncated),unit:clip(fact.numeric.unit,40,"numeric.unit",truncated),amountWon:clip(fact.numeric.amountWon,40,"numeric.amountWon",truncated)}:undefined;return {id:fact.id,key:clip(fact.key,180,"key",truncated),value,numeric,periodStart:fact.periodStart,periodEnd:fact.periodEnd,asOf:fact.asOf,state:fact.state,recognition:fact.recognition,validation:fact.validation.filter(v=>v.severity==="error").slice(0,3).map(v=>({code:text(v.code,80),severity:v.severity})),refs:fact.refs.slice(0,1).map(ref=>({documentId:ref.documentId,page:ref.page,quote:clip(ref.quote,450,"refs.quote",truncated),sourceHash:ref.sourceHash})),structuredRefs:fact.structuredRefs?.slice(0,1).map(ref=>({sourceRecordId:ref.sourceRecordId,jsonPointer:clip(ref.jsonPointer,300,"structuredRefs.jsonPointer",truncated),recordType:data.structuredTest?.sourceRecords.find(record=>record.id===ref.sourceRecordId)?.recordType})),evidenceMode:fact.structuredRefs?.length?"assumed_extracted_synthetic":"uploaded_document",truncatedFields:truncated};}
function selectedFacts(input:LegalReviewPacketInput){
  const all=input.facts,selected=new Map<string,Fact>(),counts:Record<string,{total:number;provided:number;omitted:number;axisFallbackTotal:number;axisFallbackProvided:number}>={};
  for(const check of input.checks){let checkCount=0;for(const feature of check.features){const key=`${check.id}:${feature}`,matches=all.map(fact=>({fact,tier:featureMatchTier(feature,fact)})).filter((item):item is {fact:Fact;tier:0|1|2}=>item.tier!==null).sort((a,b)=>a.tier-b.tier||priority(a.fact)-priority(b.fact));const specific=matches.filter(item=>item.tier<2).map(item=>item.fact),fallback=matches.filter(item=>item.tier===2).map(item=>item.fact),limit=Math.min(MAX_FACTS_PER_FEATURE,MAX_FACTS_PER_CHECK-checkCount),picked=specific.length?specific.slice(0,Math.max(0,limit)):fallback.slice(0,Math.max(0,limit));counts[key]={total:specific.length,provided:specific.slice(0,limit).length,omitted:Math.max(0,specific.length-limit),axisFallbackTotal:fallback.length,axisFallbackProvided:specific.length?0:picked.length};for(const fact of picked){selected.set(fact.id,fact);checkCount++;}}}
  // Risk feature overlap is expected. Add a small shared evidence margin, never an unbounded fact prefix.
  const ranked=[...selected.values()].sort((a,b)=>priority(a)-priority(b));const relevantIds=new Set(all.filter(fact=>input.checks.some(check=>check.features.some(feature=>factMatches(feature,fact)))).map(fact=>fact.id));return {facts:ranked.slice(0,MAX_FACTS),featureCounts:counts,totalRelevantFeatureFacts:relevantIds.size};
}
function compactFinding(finding:Finding){return {id:finding.id,category:finding.category,severity:finding.severity,outcome:finding.outcome,title:text(finding.title,180),detail:text(finding.detail,500),objectId:finding.objectId,factIds:finding.factIds.slice(0,12),ruleSourceIds:finding.ruleSourceIds.slice(0,8),nextAction:text(finding.nextAction,300)};}
function transactionSummary(review:any){
  if(!review||typeof review!=="object")return {provided:false};
  const items=Array.isArray(review.items)?review.items:[],candidates=Array.isArray(review.transferCandidates)?review.transferCandidates:[],coverage=review.coverage??{},totals=review.totals??{};
  const includedItems=items.slice(0,12),includedCandidates=candidates.slice(0,8);
  return {provided:true,policy:review.policy?{id:text(review.policy.id,100),title:text(review.policy.title,180),legalThreshold:review.policy.legalThreshold,automaticAdjudication:review.policy.automaticAdjudication}:undefined,itemCounts:Object.fromEntries(["pass","fail","missing","conflict","not_applicable"].map(status=>[status,items.filter((item:any)=>item.status===status).length])),items:includedItems.map((item:any)=>({id:text(item.id,100),status:item.status,title:text(item.title,160),reason:text(item.reason,400),factIds:Array.isArray(item.factIds)?item.factIds.slice(0,10):[]})),totalItems:items.length,omittedItems:Math.max(0,items.length-includedItems.length),truncatedItemIds:includedItems.filter((item:any)=>item.title.length>160||item.reason.length>400).map((item:any)=>item.id),candidateCount:candidates.length,candidates:includedCandidates.map((item:any)=>({id:text(item.id,100),status:item.status,reason:text(item.reason,300),factIds:Array.isArray(item.factIds)?item.factIds.slice(0,10):[]})),omittedCandidates:Math.max(0,candidates.length-includedCandidates.length),truncatedCandidateIds:includedCandidates.filter((item:any)=>item.reason.length>300).map((item:any)=>item.id),coverage:{state:coverage.state,transactionCount:coverage.transactionCount,inspectedCount:coverage.inspectedCount,usableAmountAndSourceCount:coverage.usableAmountAndSourceCount,unresolvedRowCount:coverage.unresolvedRowCount,completePeriodProven:coverage.completePeriodProven===true,firstObservedDate:coverage.firstObservedDate,lastObservedDate:coverage.lastObservedDate,note:text(coverage.note,300)},totals:{scope:totals.scope,inputComplete:totals.inputComplete,inflowWon:totals.inflowWon,outflowWon:totals.outflowWon,netFlowWon:totals.netFlowWon,note:text(totals.note,400)},complete:false,absenceIsNotNegativeEvidence:true};
}
function compactRule(rule:any){const truncated:string[]=[];return {id:rule.id,title:clip(rule.title,220,"title",truncated),url:clip(rule.url??rule.source_url,500,"url",truncated),article:clip(rule.article,100,"article",truncated),effectiveFrom:rule.effective_date??rule.effectiveFrom,authorityScope:clip(rule.authority_scope,180,"authorityScope",truncated),condition:clip(rule.condition,600,"condition",truncated),effect:clip(rule.effect,600,"effect",truncated),truncatedFields:truncated};}
function compactGuidance(item:any){const truncated:string[]=[];const list=(values:unknown[],max:number,key:string)=>{if(values.length>max)truncated.push(key);return values.slice(0,max).map(value=>typeof value==="string"?clip(value,160,key,truncated):value);};const calc=JSON.stringify(item.calculation??"");return {id:clip(item.id,100,"id",truncated),trigger:clip(item.trigger,300,"trigger",truncated),documents:Array.isArray(item.documents)?list(item.documents,8,"documents"):[],features:Array.isArray(item.features)?list(item.features,12,"features"):[],ruleIds:Array.isArray(item.ruleIds)?list(item.ruleIds,12,"ruleIds"):[],calculation:clip(calc,500,"calculation",truncated),outputTargets:Array.isArray(item.outputTargets)?list(item.outputTargets,8,"outputTargets"):[],missingAction:clip(item.missingAction,400,"missingAction",truncated),decisionBoundary:clip(item.decisionBoundary,500,"decisionBoundary",truncated),implementationGap:clip(item.implementationGap,500,"implementationGap",truncated),truncatedFields:truncated};}
function compactRisk(check:LegalReviewRisk,providedRuleIds:Set<string>){const truncated:string[]=[];if(check.features.length>20)truncated.push("features");return {id:check.id,title:clip(check.title,220,"title",truncated),conditions:clip(check.conditions,900,"conditions",truncated),features:check.features.slice(0,20),request:clip(check.request,800,"request",truncated),legalRuleIds:check.legalRuleIds.filter(id=>providedRuleIds.has(id)),manual:check.manual,truncatedFields:truncated};}

  const findingsByRisk=input.checks.map(check=>{const evidenceIds=new Set(input.facts.filter(fact=>check.features.some(feature=>featureMatchTier(feature,fact)!==null&&featureMatchTier(feature,fact)!==2)).map(fact=>fact.id));return input.findings.filter(finding=>finding.outcome!=="pass"&&finding.factIds.some(id=>evidenceIds.has(id))).slice(0,3);});

  const legalIds=new Set(input.checks.flatMap(check=>check.legalRuleIds));
  const allRules=input.rules.filter(rule=>legalIds.has(rule.id));

  const guidance=input.guidance.slice(0,8).map(compactGuidance);

  return {payload,text:serialized,providedFactIds:payload.providedFacts.map((fact:any)=>fact.id),coverage:payload.evidenceCoverage};
}

export interface AdvisoryCheck {id:string;status:string;reason:string;factIds:string[];ruleIds:string[];missingDocuments:string[];nextAction:string}
export function currentCompletedLegalCheckIds(rows:Array<{payload:string}>,receipt:{inputSha256:string;referenceSha256:string;contractVersion:string},expectedIds:Set<string>):Set<string>{
  const statuses=new Set(["pass","fail","missing","conflict","not_applicable","legal_review"]),completed=new Set<string>();
  for(const row of rows){try{const payload=JSON.parse(row.payload),execution=payload?.calculatorExecution;if(execution?.inputSha256!==receipt.inputSha256||execution?.referenceRelease?.sha256!==receipt.referenceSha256||execution?.contractVersion!==receipt.contractVersion||!Array.isArray(payload.checks))continue;
    for(const check of payload.checks)if(check&&expectedIds.has(check.id)&&statuses.has(check.status)&&Array.isArray(check.factIds)&&Array.isArray(check.ruleIds)&&Array.isArray(check.missingDocuments)&&typeof check.reason==="string"&&typeof check.nextAction==="string")completed.add(check.id);
  }catch{/* malformed historical rows do not suppress a current review */}}
  return completed;
}
export function gateLegalReviewChecks(input:{batch:LegalReviewRisk[];response:AdvisoryCheck[];providedFacts:Fact[];eligibleFactIds:Set<string>;validRuleIds:Set<string>;coverage:Record<string,any>}):AdvisoryCheck[]{
  const ids=new Set(input.batch.map(check=>check.id));
  // The caller derives this allowlist from source hash / structured-record and lineage validation.
  // Never infer source eligibility from model-provided identifiers or a nonempty refs array here.
  const validFacts=new Set(input.providedFacts.filter(fact=>input.eligibleFactIds.has(fact.id)&&fact.state!=="disputed"&&fact.state!=="retracted"&&fact.recognition!=="rejected").map(fact=>fact.id));
  const clearable=new Set(input.providedFacts.filter(fact=>validFacts.has(fact.id)&&fact.state==="verified"&&fact.recognition==="accepted"&&!fact.validation.some(v=>v.severity!=="info")).map(fact=>fact.id));
  const checked=Object.fromEntries(input.response.filter(check=>ids.has(check.id)).map(check=>[check.id,{...check}]));
  return input.batch.map(check=>{
    const item=checked[check.id]??{id:check.id,status:"missing",reason:"이 항목의 검토 응답이 누락되었습니다.",factIds:[],ruleIds:[],missingDocuments:[check.request],nextAction:"해당 항목을 다시 검토하세요."};
    item.factIds=item.factIds.filter(id=>validFacts.has(id));item.ruleIds=item.ruleIds.filter(id=>input.validRuleIds.has(id));
    const featureCoverageComplete=check.features.length<=20&&check.features.every(feature=>{const covered=input.coverage.featureCounts?.[`${check.id}:${feature}`];return !!covered&&covered.total>0&&covered.provided===covered.total&&covered.omitted===0&&covered.axisFallbackProvided===0;});
    const ruleCoverage=input.coverage.checkRules?.[check.id],ruleCoverageComplete=!!ruleCoverage&&ruleCoverage.omitted===0;
    const guidanceCoverage=input.coverage.checkGuidance?.[check.id],guidanceCoverageComplete=!!guidanceCoverage&&guidanceCoverage.omitted===0;
    const clippedFacts=(input.coverage.truncatedFactIdsByCheck?.[check.id]??[]).length>0,clippedRules=(input.coverage.truncatedRuleIdsByCheck?.[check.id]??[]).length>0,clippedGuidance=(input.coverage.truncatedGuidanceIdsByCheck?.[check.id]??[]).length>0,clippedCheck=(input.coverage.truncatedCheckIds??[]).includes(check.id);
    const axes=new Set(check.features.map(axis)),inventoryCoverage=input.coverage.inventoryRowsOmitted??{};
    const inventoryKey:Record<string,string>={income:"incomes",debt:"debts",asset:"assets",transaction:"transactions"};
    const relevantInventoryKinds=["income","debt","asset","transaction"].filter(kind=>axes.has(kind));
    const inventoryCoverageComplete=relevantInventoryKinds.every(kind=>(inventoryCoverage[inventoryKey[kind]]??0)===0);
    const transactionCoverage=input.coverage.transactionReview??{},transactionCoverageComplete=!axes.has("transaction")||((transactionCoverage.omittedItems??0)===0&&(transactionCoverage.omittedCandidates??0)===0&&transactionCoverage.completePeriodProven===true&&(transactionCoverage.unresolvedRowCount??1)===0&&(input.coverage.truncatedTransactionItemIds??[]).length===0&&(input.coverage.truncatedTransactionCandidateIds??[]).length===0);
    if(["pass","not_applicable"].includes(item.status)&&(!featureCoverageComplete||!ruleCoverageComplete||!guidanceCoverageComplete||!inventoryCoverageComplete||!transactionCoverageComplete||clippedFacts||clippedRules||clippedGuidance||clippedCheck)){item.status="missing";item.reason+=" / 이 체크의 관련 원문 사실·규칙·예외 가이드·목록 중 일부 생략 또는 문구 절단이 있어 통과·비해당으로 확정하지 않았습니다.";}
    const citedClearableFacts=input.providedFacts.filter(fact=>item.factIds.includes(fact.id)&&clearable.has(fact.id));
    const missingFeatureCitation=!check.manual&&check.features.some(feature=>!citedClearableFacts.some(fact=>{const tier=featureMatchTier(feature,fact);return tier!==null&&tier<2;}));
    if(["pass","not_applicable"].includes(item.status)&&(!item.factIds.length||item.factIds.some(id=>!clearable.has(id))||missingFeatureCitation)){item.status="missing";item.reason+=" / 제공되고 검증·인정된 사실근거가 없거나 각 필수 피처의 인용 근거가 빠져 해소하지 않았습니다.";}
    if(check.manual&&["pass","not_applicable"].includes(item.status))item.status="legal_review";
    if(item.status==="not_applicable"&&!input.providedFacts.some(fact=>item.factIds.includes(fact.id)&&[false,"no","없음"].includes(fact.value as string|boolean)&&check.features.some(key=>axis(key)===axis(fact.key)))){item.status="legal_review";item.reason+=" / 제공된 해당 조건의 명시적 음성 증거가 없어 비해당으로 처리하지 않았습니다.";}
    return item;
  });
